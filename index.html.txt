  function renderOverview(c){
    const r=records();
    const ds=dates();
    const ns=names();

    const now=new Date();
    const todayKey=new Date(now.getFullYear(),now.getMonth(),now.getDate()).getTime();
    const weekStart=new Date(now);
    const day=(weekStart.getDay()+6)%7; // Monday=0
    weekStart.setHours(0,0,0,0);
    weekStart.setDate(weekStart.getDate()-day);
    const monthStart=new Date(now.getFullYear(),now.getMonth(),1);
    const nextMonth=new Date(now.getFullYear(),now.getMonth()+1,1);

    function recordTime(rec){
      if(!rec) return 0;
      const t=Date.parse(rec.completedAt||"");
      return Number.isFinite(t)?t:0;
    }
    function inWeek(rec){
      const t=recordTime(rec);
      return t>=weekStart.getTime();
    }
    function inMonth(rec){
      const t=recordTime(rec);
      return t>=monthStart.getTime() && t<nextMonth.getTime();
    }
    function isToday(rec){
      const t=recordTime(rec);
      if(!t)return false;
      const d=new Date(t);
      return new Date(d.getFullYear(),d.getMonth(),d.getDate()).getTime()===todayKey;
    }
    function avg(list){
      const scores=list.map(x=>Number(x.score)).filter(Number.isFinite);
      return scores.length?(scores.reduce((a,b)=>a+b,0)/scores.length).toFixed(1):"-";
    }
    function maxScore(list){
      const scores=list.map(x=>Number(x.score)).filter(Number.isFinite);
      return scores.length?Math.max.apply(null,scores):"-";
    }
    function allAssigned(){
      const rows=[];
      ds.forEach(d=>ns.forEach(n=>{
        const rk=rankOf(d,n);
        if(rk) rows.push({d,n,rec:r[d+"__"+n]});
      }));
      return rows;
    }

    const assigned=allAssigned();
    const allRecs=assigned.filter(x=>x.rec).map(x=>x.rec);
    const todayRecs=allRecs.filter(isToday);
    const weekRecs=allRecs.filter(inWeek);
    const monthRecs=allRecs.filter(inMonth);

    const todayAssigned=assigned.filter(x=>isToday(x.rec) || (x.d===new Date(now.getTime()-now.getTimezoneOffset()*60000).toISOString().slice(0,10)));
    const todayDone=todayRecs.length;
    const todayTarget=todayAssigned.length;

    function statCard(title,value,sub){
      return `<div style="flex:1;min-width:145px;background:#f7f8fa;border-radius:16px;padding:16px">
        <div style="font-size:13px;color:#667085">${title}</div>
        <div style="font-size:28px;font-weight:900;margin:4px 0">${value}</div>
        <div style="font-size:12px;color:#667085">${sub||""}</div>
      </div>`;
    }

    const personRows=ns.map(n=>{
      const recs=assigned.filter(x=>x.n===n&&x.rec).map(x=>x.rec);
      const thisWeek=recs.filter(inWeek);
      const thisMonth=recs.filter(inMonth);
      return {
        n,
        total:recs.length,
        totalAvg:avg(recs),
        week:thisWeek.length,
        weekAvg:avg(thisWeek),
        month:thisMonth.length,
        monthAvg:avg(thisMonth)
      };
    });

    const nonRespondersToday=ns.filter(n=>{
      const hasToday=assigned.some(x=>{
        const d=new Date(x.d+"T12:00:00");
        const same=d.getFullYear()===now.getFullYear() &&
                   d.getMonth()===now.getMonth() &&
                   d.getDate()===now.getDate();
        return same && x.n===n && x.rec;
      });
      const assignedToday=assigned.some(x=>{
        const d=new Date(x.d+"T12:00:00");
        return d.getFullYear()===now.getFullYear() &&
               d.getMonth()===now.getMonth() &&
               d.getDate()===now.getDate() &&
               x.n===n;
      });
      return assignedToday && !hasToday;
    });

    const dateRows=ds.map(d=>{
      const rr=ns.map(n=>({n,rec:r[d+"__"+n],rk:rankOf(d,n)})).filter(x=>x.rk);
      const done=rr.filter(x=>x.rec);
      const scores=done.map(x=>Number(x.rec.score)).filter(Number.isFinite);
      return {
        d,
        target:rr.length,
        done:done.length,
        avg:scores.length?(scores.reduce((a,b)=>a+b,0)/scores.length).toFixed(1):"-",
        max:scores.length?Math.max.apply(null,scores):"-"
      };
    });

    c.innerHTML=`
      <h3>📊 퀴즈 현황</h3>
      <p class="small">공용 저장소에 누적된 응시 기록을 기준으로 자동 집계됩니다.</p>

      <div style="display:flex;gap:10px;flex-wrap:wrap;margin:12px 0 18px">
        ${statCard("오늘 응시",todayDone,"오늘 배정 "+todayTarget+"명")}
        ${statCard("오늘 평균",avg(todayRecs),"오늘 응시 기준")}
        ${statCard("이번 주 응시",weekRecs.length,"월요일부터 현재까지")}
        ${statCard("이번 주 평균",avg(weekRecs),"이번 주 응시 기준")}
        ${statCard("이번 달 응시",monthRecs.length,"이번 달 누적")}
        ${statCard("이번 달 평균",avg(monthRecs),"이번 달 응시 기준")}
      </div>

      <div style="display:flex;gap:10px;flex-wrap:wrap;margin-bottom:16px">
        ${statCard("누적 응시",allRecs.length,"등록된 전체 기록")}
        ${statCard("누적 평균",avg(allRecs),"전체 응시 기준")}
        ${statCard("누적 최고점",maxScore(allRecs),"전체 응시 기준")}
        ${statCard("오늘 미응시",nonRespondersToday.length,"오늘 배정자 중 미응시")}
      </div>

      <div style="margin-top:14px;padding:14px;border:1px solid #eee;border-radius:16px;background:#fff;overflow:auto">
        <h4 style="margin:0 0 10px">📅 날짜별 현황</h4>
        <table style="width:100%;border-collapse:collapse;min-width:520px">
          <thead><tr>
            <th style="padding:9px;border-bottom:1px solid #ddd;text-align:left">교육일</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">대상</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">응시</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">미응시</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">평균</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">최고점</th>
          </tr></thead>
          <tbody>
            ${dateRows.map(x=>`<tr>
              <td style="padding:9px;border-bottom:1px solid #eee">${esc(labels[x.d])}</td>
              <td style="padding:9px;text-align:center;border-bottom:1px solid #eee">${x.target}</td>
              <td style="padding:9px;text-align:center;border-bottom:1px solid #eee">✅ ${x.done}</td>
              <td style="padding:9px;text-align:center;border-bottom:1px solid #eee">⏳ ${x.target-x.done}</td>
              <td style="padding:9px;text-align:center;border-bottom:1px solid #eee">${x.avg}</td>
              <td style="padding:9px;text-align:center;border-bottom:1px solid #eee">${x.max}</td>
            </tr>`).join("")}
          </tbody>
        </table>
      </div>

      <div style="margin-top:14px;padding:14px;border:1px solid #eee;border-radius:16px;background:#fff;overflow:auto">
        <h4 style="margin:0 0 10px">👥 드리미별 누적 현황</h4>
        <table style="width:100%;border-collapse:collapse;min-width:650px">
          <thead><tr>
            <th style="padding:9px;border-bottom:1px solid #ddd;text-align:left">드리미</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">누적 응시</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">누적 평균</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">이번 주 응시</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">이번 주 평균</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">이번 달 응시</th>
            <th style="padding:9px;border-bottom:1px solid #ddd">이번 달 평균</th>
          </tr></thead>
          <tbody>
            ${personRows.map(x=>`<tr>
              <td style="padding:9px;border-bottom:1px solid #eee">${esc(x.n)}</td>
              <td style="padding:9px;text-align:center;border-bottom:1px solid #eee">${x.total}</td>

  function snapshot(){

  function readLocal(key,fallback){
    try{

      return v==null?fallback:v;

  }

  function hasObjectData(v){
    return v && typeof v==="object" && !Array.isArray(v) && Object.keys(v).length>0;
  }

  function apply(data){
    if(!data)return;

    // IMPORTANT: an empty first response from Google Apps Script must never
    // erase the existing local roster/schedule. Cloud data wins only when
    // that field actually contains data.
    if(Array.isArray(data.names) && data.names.length){
      localStorage.setItem("eunpyeongQuizNames",JSON.stringify(data.names));
    }

    if(Array.isArray(data.schedules) && data.schedules.length){
      localStorage.setItem("fireQuizSchedulesV5",JSON.stringify(data.schedules));
      localStorage.setItem("fireQuizSchedules",JSON.stringify(data.schedules));
      localStorage.setItem("eunpyeongQuizSchedules",JSON.stringify(data.schedules));
    }

    if(hasObjectData(data.records)){
      localStorage.setItem("fireQuizRecords",JSON.stringify(data.records));
    }

    localStorage.setItem(SYNCED_AT_KEY,new Date().toISOString());
  }

  function localSnapshotHasData(){
    const s=snapshot();
    return s.names.length>0 || s.schedules.length>0 || hasObjectData(s.records);
  }

  async function cloudGet(){
    const url=getUrl();
    if(!url) throw new Error("공용 저장소 URL이 등록되지 않았습니다.");
    const r=await fetch(url,{method:"GET",cache:"no-store"});
    if(!r.ok) throw new Error("공용 저장소를 불러오지 못했습니다. HTTP "+r.status);
    const j=await r.json();
    if(!j.ok) throw new Error(j.error||"공용 저장소 응답 오류");
    return j.data||{};
  }

  async function cloudPut(){
    const url=getUrl();
    if(!url) throw new Error("공용 저장소 URL이 등록되지 않았습니다.");
    const r=await fetch(url,{
      method:"POST",
      headers:{"Content-Type":"text/plain;charset=utf-8"},
      body:JSON.stringify({action:"save",data:snapshot()})
    });
    if(!r.ok) throw new Error("공용 저장소 저장 실패. HTTP "+r.status);
    const j=await r.json();
    if(!j.ok) throw new Error(j.error||"공용 저장소 저장 오류");
    localStorage.setItem(SYNCED_AT_KEY,new Date().toISOString());
    return j;
  }

<script id="v20-shared-data">

  const CLOUD_URL='https://script.google.com/macros/s/AKfycbxyECpbtn9xwI5xmfsi37-6ZFX4jFYgb2zs8qdOv0rWrCIp-SeW0HtxAvEc55mgFBeJ/exec';
  const RECORD_KEY='fireQuizRecords';
  const NAMES_KEY='eunpyeongQuizNames';
  const SCHEDULE_KEY='fireQuizSchedulesV5';
  const LOCK_KEY='fireQuizV20RecordLock';

  function readJSON(k,f){try{const v=JSON.parse(localStorage.getItem(k)||'null');return v==null?f:v}catch(e){return f}}
  function snapshot(){
    return {
      version:20,
      names:readJSON(NAMES_KEY,[]),
      schedules:readJSON(SCHEDULE_KEY,[]),
      records:readJSON(RECORD_KEY,{})
    };
  }
  function saveRecords(r){localStorage.setItem(RECORD_KEY,JSON.stringify(r||{}));}
  function time(r){const t=Date.parse(r&&r.completedAt||'');return Number.isFinite(t)?t:0;}
  function mergeRecords(cloud,local){
    const out={...(cloud||{})};
    Object.keys(local||{}).forEach(k=>{
      const l=local[k],c=out[k];
      if(!c||time(l)>=time(c))out[k]=l;
    });
    return out;
  }
  async function request(method,payload){
    const o={method,cache:'no-store'};
    if(method==='POST'){
      o.headers={'Content-Type':'text/plain;charset=utf-8'};
      o.body=JSON.stringify(payload);
    }
    const r=await fetch(CLOUD_URL,o);
    if(!r.ok)throw new Error('공용 저장소 HTTP '+r.status);
    const j=await r.json();
    if(!j.ok)throw new Error(j.error||'공용 저장소 오류');
    return j.data||{};
  }
  async function push(){return request('POST',{action:'save',data:snapshot()})}

  async function syncRecords(){
    if(localStorage.getItem(LOCK_KEY))return;
    localStorage.setItem(LOCK_KEY,'1');
    try{
      const local=snapshot();
      const cloud=await request('GET');
      const cloudRecords=(cloud.records&&typeof cloud.records==='object')?cloud.records:{};
      const merged=mergeRecords(cloudRecords,local.records);
      saveRecords(merged);

      const localHasNew=Object.keys(local.records).some(k=>!cloudRecords[k]||time(local.records[k])>time(cloudRecords[k]));
      const cloudHasNew=Object.keys(cloudRecords).some(k=>!local.records[k]||time(cloudRecords[k])>time(local.records[k]));
      if(localHasNew||cloudHasNew){
        const data={
          version:20,
          names:(Array.isArray(cloud.names)&&cloud.names.length)?cloud.names:local.names,
          schedules:(Array.isArray(cloud.schedules)&&cloud.schedules.length)?cloud.schedules:local.schedules,
          records:merged
        };
        await request('POST',{action:'save',data:data});
      }
      try{if(typeof renderStatus==='function')renderStatus()}catch(e){}
      try{if(typeof renderOverview==='function')renderOverview(document.getElementById('dashContent'))}catch(e){}
    }catch(e){console.warn('V20 record sync:',e.message)}
    finally{localStorage.removeItem(LOCK_KEY)}
  }

  function wrapSave(){
    const old=window.saveQuizRecord;
    if(typeof old!=='function'||old.__v20recordWrapped)return;
    // already wrapped by an earlier version: replace the wrapper anyway below
  }

  // saveQuizRecord is declared by the main quiz script before this block runs.
  // Always wrap it so every completed quiz is uploaded immediately.
  function install(){
    const old=window.saveQuizRecord;
    if(typeof old==='function'&&!old.__v20recordWrapped){

      };

      window.saveQuizRecord=wrapped;
    }
  }
